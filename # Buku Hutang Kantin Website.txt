<script>

let hutangData = JSON.parse(localStorage.getItem("hutangData")) || [];

function simpanData(){
  localStorage.setItem("hutangData", JSON.stringify(hutangData));
}

function suara(text){
  let speech = new SpeechSynthesisUtterance(text);
  speech.lang = "ms-MY";
  speech.volume = 1;
  speech.rate = 1;
  speech.pitch = 1;
  window.speechSynthesis.speak(speech);
}

function vibration(){
  if(navigator.vibrate){
    navigator.vibrate([300,200,300,200,500]);
  }
}

function tambahHutang(){

  let nama = document.getElementById("nama").value.trim();
  let jumlah = parseFloat(document.getElementById("jumlah").value);

  if(nama === "" || isNaN(jumlah)){
    alert("Sila isi nama dan jumlah hutang!");
    return;
  }

  let jumpa = hutangData.find(item => item.nama.toLowerCase() === nama.toLowerCase());

  if(jumpa){
    jumpa.hutang += jumlah;

    suara(nama + " hutang bertambah menjadi RM" + jumpa.hutang);

    if(jumpa.hutang >= 50){
      vibration();
      alert("⚠️ Hutang pembeli melebihi had!");
      suara("Amaran. Hutang pembeli melebihi had");
    }

  } else {

    hutangData.push({
      nama:nama,
      hutang:jumlah
    });

    suara(nama + " hutang RM" + jumlah);

    if(jumlah >= 50){
      vibration();
      alert("⚠️ Hutang pembeli melebihi had!");
      suara("Amaran. Hutang pembeli melebihi had");
    }
  }

  simpanData();
  paparData();

  document.getElementById("nama").value = "";
  document.getElementById("jumlah").value = "";
}

function bayar(index){

  let nama = hutangData[index].nama;

  suara(nama + " sudah bayar hutang");

  hutangData.splice(index,1);

  simpanData();
  paparData();
}

function padam(index){
  hutangData.splice(index,1);

  simpanData();
  paparData();
}

function paparData(){

  let list = document.getElementById("hutangList");

  list.innerHTML = "";

  hutangData.forEach((item,index)=>{

    let warningText = "";

    if(item.hutang >= 50){
      warningText = `<div class="warning">⚠️ Hutang Melebihi Had!</div>`;
    }

    list.innerHTML += `

      <div class="hutang-item">

        <div class="hutang-info">
          <h2>${item.nama}</h2>
          <p>Hutang: RM${item.hutang.toFixed(2)}</p>
          ${warningText}
        </div>

        <div class="buttons">
          <button class="paid-btn" onclick="bayar(${index})">Dah Bayar</button>

          <button class="delete-btn" onclick="padam(${index})">Padam</button>
        </div>

      </div>

    `;
  });
}

paparData();

</script>

# Cara Letak Dalam GitHub

1. Buka VS Code.
2. Buat folder baru.
3. Buat fail bernama `index.html`.
4. Copy SEMUA coding atas.
5. Paste dalam `index.html`.
6. Save.
7. Upload folder ke GitHub.
8. Pergi Settings > Pages.
9. Pilih branch `main`.
10. Tekan Save.
11. Website siap.
